\newcommand{\MCnumber}[1]{\pgfmathprintnumber[fixed,fixed zerofill,precision=3]{#1}}

% Freeze numerical values before PGFPlots defers drawing to the end of the axis.
% Arguments: y, color, marker, estimate, row low/high, MC low/high.
\newcommand{\DrawContrast}[8]{%
  \draw[abmGray,line width=0.75pt] (axis cs:#5,#1) -- (axis cs:#6,#1);%
  \draw[abmGray,line width=0.5pt]
    ([yshift=-1.2pt]axis cs:#5,#1) -- ([yshift=1.2pt]axis cs:#5,#1)
    ([yshift=-1.2pt]axis cs:#6,#1) -- ([yshift=1.2pt]axis cs:#6,#1);%
  \addplot[only marks,mark=#3,mark size=1.55pt,#2,forget plot]
    coordinates {(#4,#1)};%
  \node[font=\fontsize{8}{10}\selectfont,text=black!85]
    at ([xshift=1.275cm]axis cs:0.6,#1)
    {[\MCnumber{#7},\,\MCnumber{#8}]};%
}

% Arguments: CSV row (zero-based), vertical position, actor color, marker.
\newcommand{\ContrastRow}[4]{%
  \pgfplotstablegetelem{#1}{estimate}\of{\HtwoData}\edef\PointEstimate{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{row_ci95_low}\of{\HtwoData}\edef\RowLow{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{row_ci95_high}\of{\HtwoData}\edef\RowHigh{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{mc_ci95_low}\of{\HtwoData}\edef\MCLow{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{mc_ci95_high}\of{\HtwoData}\edef\MCHigh{\pgfplotsretval}%
  \edef\DrawThisRow{\noexpand\DrawContrast{#2}{#3}{#4}{\PointEstimate}
    {\RowLow}{\RowHigh}{\MCLow}{\MCHigh}}%
  \DrawThisRow
}
