% Freeze values before PGFPlots defers drawing.
% Arguments: y, color, marker shape, estimate, CI lower, CI upper.
\newcommand{\DrawOutcome}[6]{%
  \addplot[only marks,mark=#3,mark size=\PointSize,
    mark options={draw=#2,fill=#2,solid},forget plot]
    coordinates {(#4,#1)};
  % All intervals sit in a fixed column, separate from the plot area.
  \node[anchor=center,text=black,inner sep=0pt,
    font=\fontsize{8.5}{10}\selectfont]
    at ([xshift=1.15cm]axis cs:80,#1)
    {$[\pgfmathprintnumber[fixed,fixed zerofill,precision=2]{#5},\,
       \pgfmathprintnumber[fixed,fixed zerofill,precision=2]{#6}]$};
}
% Arguments: zero-based CSV row, y, color, marker shape.
\newcommand{\OutcomeRow}[4]{%
  \pgfplotstablegetelem{#1}{estimate}\of{\HthreeData}%
  \edef\ThisEstimate{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{ci_low}\of{\HthreeData}%
  \edef\ThisLow{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{ci_high}\of{\HthreeData}%
  \edef\ThisHigh{\pgfplotsretval}%
  \edef\DrawThis{\noexpand\DrawOutcome{#2}{#3}{#4}{\ThisEstimate}{\ThisLow}{\ThisHigh}}%
  \DrawThis
}
