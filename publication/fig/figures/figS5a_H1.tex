\newcommand{\DrawContrast}[6]{%
  \addplot[only marks,mark=#3,mark size=\PointSize,
    mark options={draw=#2,fill=#2,solid},forget plot]
    coordinates {(#4,#1)};
  \node[anchor=center,text=black,inner sep=0pt,
    font=\fontsize{8.5}{10}\selectfont]
    at ([xshift=1.15cm]axis cs:2,#1)
    {$[\pgfmathprintnumber[fixed,fixed zerofill,precision=2]{#5},\,
       \pgfmathprintnumber[fixed,fixed zerofill,precision=2]{#6}]$};
}
% Arguments: zero-based CSV row, y, color, marker.
% Freeze numeric values before PGFPlots defers drawing.
\newcommand{\ContrastRow}[4]{%
  \pgfplotstablegetelem{#1}{estimate}\of{\HoneData}%
  \edef\ThisEstimate{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{ci_low}\of{\HoneData}%
  \edef\ThisLow{\pgfplotsretval}%
  \pgfplotstablegetelem{#1}{ci_high}\of{\HoneData}%
  \edef\ThisHigh{\pgfplotsretval}%
  \edef\DrawThis{\noexpand\DrawContrast{#2}{#3}{#4}{\ThisEstimate}{\ThisLow}{\ThisHigh}}%
  \DrawThis
}
