import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns

# Set global style
sns.set_theme(style="whitegrid")
plt.rcParams.update({
    'font.family': 'Times New Roman',
    'font.size': 10,
    'axes.labelsize': 10,
    'axes.titlesize': 10,
    'xtick.labelsize': 9,
    'ytick.labelsize': 9
})

# Data
pollution_levels = [10, 30, 50]
titles = ['Low Pollution (10%)', 'Medium Pollution (30%)', 'High Pollution (50%)']
no_treatment = [14.943, 8.575, 5.240]

accuracy_values = [70, 80, 90]
accuracy_effect = [
    [32.131, 34.238, 36.671],
    [26.281, 28.678, 31.077],
    [23.159, 25.849, 27.815]
]

speed_values = [50, 75, 100]
speed_effect = [
    [28.053, 34.238, 40.456],
    [22.420, 28.678, 34.935],
    [18.889, 25.849, 32.257]
]

# Create figure
fig, axes = plt.subplots(3, 1, figsize=(5.5, 12), sharex=False, sharey=True)

for i, pollution in enumerate(pollution_levels):
    ax = axes[i]

    # Accuracy line
    ax.plot(accuracy_values, accuracy_effect[i], 'o-',
            color='#1f77b4', linewidth=1.5, markersize=5,
            label='platform-accuracy (speed=75)')

    # Speed line
    ax.plot(speed_values, speed_effect[i], 's-',
            color='#ff7f0e', linewidth=1.5, markersize=5,
            label='platform-speed (accuracy=80)')

    # Reference line
    ax.axhline(y=no_treatment[i], linestyle='--', color='gray',
               label='no treatment')

    # Accuracy slope annotations
    for j in range(len(accuracy_values) - 1):
        slope = (accuracy_effect[i][j+1] - accuracy_effect[i][j]) / (accuracy_values[j+1] - accuracy_values[j])
        norm_slope = slope * (accuracy_values[j] / accuracy_effect[i][j])
        ax.annotate(f'S={norm_slope:.2f}',
                    xy=((accuracy_values[j]+accuracy_values[j+1])/2,
                        (accuracy_effect[i][j]+accuracy_effect[i][j+1])/2),
                    xytext=(0, -12),
                    textcoords='offset points',
                    color='#1f77b4', backgroundcolor='white', fontsize=8)

    # Speed slope annotations
    for j in range(len(speed_values) - 1):
        slope = (speed_effect[i][j+1] - speed_effect[i][j]) / (speed_values[j+1] - speed_values[j])
        norm_slope = slope * (speed_values[j] / speed_effect[i][j])
        ax.annotate(f'S={norm_slope:.2f}',
                    xy=((speed_values[j]+speed_values[j+1])/2,
                        (speed_effect[i][j]+speed_effect[i][j+1])/2),
                    xytext=(-17, 13),
                    textcoords='offset points',
                    color='#ff7f0e', backgroundcolor='white', fontsize=8)

    # Axis labels
    ax.set_ylabel('Effectiveness')
    ax.set_xlabel('platform-accuracy')
    ax.set_xticks(accuracy_values)
    ax.set_xticklabels(accuracy_values)

    # Twin axis for platform-speed
    ax2 = ax.twiny()
    ax2.set_xlim(ax.get_xlim())
    ax2.set_xticks(speed_values)
    ax2.set_xticklabels(speed_values)
    ax2.set_xlabel('platform-speed')

    # Add subfigure label below
    ax.text(0.5, -0.24, f'({chr(97+i)}) {titles[i]}',
            transform=ax.transAxes, ha='center', va='top',
            fontsize=9, fontweight='bold')

# Layout adjustments
plt.subplots_adjust(hspace=0.6)

# Global legend
handles, labels = axes[0].get_legend_handles_labels()
fig.legend(handles, labels,
           loc='upper center', bbox_to_anchor=(0.5, 0.99), ncol=1, frameon=False,
           fontsize=9, title='Legend (S = normalized sensitivity coefficient)',
           title_fontsize=9)

# Save to PDF
plt.savefig('fig3_platform_sensitivity_vertical.pdf', format='pdf', dpi=600, bbox_inches='tight')
plt.show()
plt.close()
print('fig3的pdf成功输出')
