# 改用 Matplotlib 内建的 mathtext 渲染（不依赖外部 LaTeX 安装）
import matplotlib.pyplot as plt
import pandas as pd

# 使用 mathtext 渲染符号（如 L, τ_L, α, n_p）
plt.rcParams.update({
    "font.family": "Times New Roman",
    "text.usetex": False,  # 关闭 LaTeX 渲染
    "font.size": 10
})

# 表格内容准备（mathtext 内嵌符号）
data = {
    "Treatment": ["Legal", "Platform"],
    "Parameter 1": ["Legal Strength $L$", "Platform Accuracy $\\alpha$"],
    "Parameter 2": ["Legal Response Time $\\tau_L$", "Platform Speed $n_p$"]
}

df = pd.DataFrame(data)

# 绘图
fig, ax = plt.subplots(figsize=(6, 1.5))
ax.axis('off')

# 创建表格
table = ax.table(cellText=df.values,
                 colLabels=df.columns,
                 cellLoc='center',
                 loc='center')

# 样式调整
table.auto_set_font_size(False)
table.set_fontsize(10)
table.scale(1.2, 1.5)

# 保存图像
plt.savefig("table2_treatment_parameters_fixed.pdf", bbox_inches='tight', dpi=600)
plt.show()
