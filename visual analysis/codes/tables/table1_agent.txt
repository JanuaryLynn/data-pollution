import matplotlib.pyplot as plt
import pandas as pd

# 数据准备
data = {
    "Class (flag)": ["User (super-spreader? = false)",
                     "Super-Spreader (super-spreader? = true)"],
    "Share in\nPopulation": ["99 %", "1 %"],
    "Visual\nSize": [0.5, 1.0],
    "Spread\nProbability": [0.10, 0.15],
    "State\nVariable": ["polluted?", "polluted?"]
}

df = pd.DataFrame(data)

# 绘图
fig, ax = plt.subplots(figsize=(8, 2))  # 适当提高高度用于第一行
ax.axis('off')

# 创建表格
table = ax.table(cellText=df.values,
                 colLabels=df.columns,
                 cellLoc='center',
                 loc='center')

# 样式调整
table.auto_set_font_size(False)
table.set_fontsize(13.5)

# 缩放：Class列保持宽度，其他列压缩，第一行高一点
for i in range(len(df.columns)):
    for j in range(len(df) + 1):  # 包括列名行
        cell = table[j, i]
        if j == 0:
            cell.set_height(0.5)  # 列名行高
        else:
            cell.set_height(0.3)
        if i == 0:
            cell.set_width(0.5)  # Class列宽度保持
        else:
            cell.set_width(0.25)  # 其他列窄一些

# 保存图像
plt.savefig("table1_agent_specification_adjusted.pdf", bbox_inches='tight',\
 dpi=600)
plt.show()
