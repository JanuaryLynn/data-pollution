import matplotlib.pyplot as plt
import numpy as np
from matplotlib.colors import LinearSegmentedColormap
import matplotlib.gridspec as gridspec

plt.rcParams.update({
    'font.family': 'Times New Roman',
    'font.size': 10,
    'axes.labelsize': 10,
    'axes.titlesize': 10,
    'xtick.labelsize': 9,
    'ytick.labelsize': 9
})

# 数据
pollution_levels = [10, 30, 50]
titles = ['Low Pollution (10%)', 'Medium Pollution (30%)', 'High Pollution (50%)']
legal_combinations = ['L1\n(S=30,R=5)', 'L2\n(S=60,R=5)', 'L3\n(S=90,R=5)', 'L4\n(S=60,R=1)', 'L5\n(S=60,R=10)']
platform_combinations = ['P1\n(A=70,S=75)', 'P2\n(A=80,S=75)', 'P3\n(A=90,S=75)', 'P4\n(A=80,S=50)', 'P5\n(A=80,S=100)']

legal_effectiveness = np.array([
    [18.273, 10.976, 7.314],
    [22.399, 13.875, 10.003],
    [28.453, 16.557, 12.524],
    [99.459, 39.492, 32.820],
    [19.477, 11.249, 7.594]
])
platform_effectiveness = np.array([
    [32.131, 26.281, 23.159],
    [34.238, 28.678, 25.849],
    [36.671, 31.077, 27.815],
    [28.053, 22.420, 18.889],
    [40.456, 34.935, 32.257]
])

advantage_matrix = np.zeros((5, 5, 3))
for i in range(5):
    for j in range(5):
        for k in range(3):
            advantage_matrix[i, j, k] = legal_effectiveness[i, k] - platform_effectiveness[j, k]

# 自定义配色
colors = ["#2166ac", "#4393c3", "#92c5de", "#d1e5f0", "#f7f7f7",
          "#fddbc7", "#f8a98e", "#f4977c", "#ef6c61"]
custom_cmap = LinearSegmentedColormap.from_list("custom", colors, N=256)

# 布局：一行三图 + 一列 colorbar
fig = plt.figure(figsize=(11, 4.5))
gs = gridspec.GridSpec(1, 4, width_ratios=[5, 5, 5, 0.25], wspace=0.4)

ims = []
for p, pollution in enumerate(pollution_levels):
    ax = fig.add_subplot(gs[0, p])
    im = ax.imshow(advantage_matrix[:, :, p], cmap=custom_cmap, vmin=-50, vmax=50, aspect='equal')
    ims.append(im)

    # 数值标签
    for i in range(5):
        for j in range(5):
            value = advantage_matrix[i, j, p]
            text_color = 'white' if abs(value) > 30 else 'black'
            ax.text(j, i, f'{value:.1f}', ha='center', va='center',
                    color=text_color, fontsize=8, fontweight='bold')

    # 星标
    best_legal = np.argmax(legal_effectiveness[:, p])
    best_platform = np.argmax(platform_effectiveness[:, p])
    best_value = advantage_matrix[best_legal, best_platform, p]
    star_color = 'white' if abs(best_value) > 30 else 'black'
    ax.plot(best_platform - 0.3, best_legal - 0.3, marker='*', markersize=7, color=star_color)

    # 标签
    ax.set_xticks(np.arange(5))
    ax.set_yticks(np.arange(5))
    ax.set_xticklabels(platform_combinations, rotation=45, ha='right')
    ax.set_yticklabels(legal_combinations if p == 0 else [])
    ax.set_xlabel('Platform Self-regulation Strategies')
    if p == 0:
        ax.set_ylabel('Legal Intervention Strategies')

    # 子图标题
    ax.text(0.5, 1.08, f'({chr(97+p)}) {titles[p]}',
            transform=ax.transAxes, ha='center', va='bottom',
            fontsize=9, fontweight='bold')

# —— 全局 colorbar，高度与子图矩阵一致 —— 
pos = ax.get_position()  # 最后一个子图的位置
first_pos = fig.axes[0].get_position()  # 第一个子图位置
last_pos = fig.axes[2].get_position()   # 第三个子图位置

cax = fig.add_axes([last_pos.x1 + 0.01, first_pos.y0, 0.01, first_pos.y1 - first_pos.y0])
cbar = fig.colorbar(ims[0], cax=cax)
cbar.set_ticks([-50, -25, 0, 25, 50])
cbar.set_label('Advantage (Legal − Platform)', fontsize=9)

# 说明文字
fig.text(0.5, -0.12, '* Indicates optimal parameter combination for each strategy\n'
         'Positive values (red): Legal intervention advantage   '
         'Negative values (blue): Platform self-regulation advantage',
         ha='center', fontsize=9)

plt.savefig('fig6_strategy_context_matrix_horizontal_equalcbar.pdf', format='pdf', dpi=600, bbox_inches='tight')
plt.show()
plt.close()
print('横向一行三图 + 子图等高 colorbar 输出成功')
