import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns

# —— 全局样式（紧凑、IEEE 友好）——
sns.set_theme(style="whitegrid")
plt.rcParams.update({
    'font.family': 'Times New Roman',
    'font.size': 9.5,
    'axes.labelsize': 9.5,
    'axes.titlesize': 9.5,
    'xtick.labelsize': 9,
    'ytick.labelsize': 9
})

# —— 数据 —— 
pollution_levels = [10, 30, 50]
titles = ['Low Pollution (10%)', 'Medium Pollution (30%)', 'High Pollution (50%)']
no_treatment = [14.943, 8.575, 5.240]

accuracy_values = [70, 80, 90]
accuracy_effect = [
    [32.131, 34.238, 36.671],
    [26.281, 28.678, 31.077],
    [23.159, 25.849, 27.815]
]

speed_values = [50, 75, 100]
speed_effect = [
    [28.053, 34.238, 40.456],
    [22.420, 28.678, 34.935],
    [18.889, 25.849, 32.257]
]

# —— 横向一行三图 —— 
fig, axes = plt.subplots(1, 3, figsize=(11, 3.4), sharex=False, sharey=True)
plt.subplots_adjust(wspace=0.25)  # 子图之间更紧凑

for i, pollution in enumerate(pollution_levels):
    ax = axes[i]

    # 折线
    ax.plot(accuracy_values, accuracy_effect[i], 'o-', color='#1f77b4',
            linewidth=1.4, markersize=4.5,
            label='platform-accuracy (speed=75)')
    ax.plot(speed_values, speed_effect[i], 's-', color='#ff7f0e',
            linewidth=1.4, markersize=4.5,
            label='platform-speed (accuracy=80)')

    # baseline
    ax.axhline(y=no_treatment[i], linestyle='--', color='gray', linewidth=1.0,
               label='no treatment')

    # 标注标准化敏感度 S
    for j in range(len(accuracy_values) - 1):
        slope = (accuracy_effect[i][j+1] - accuracy_effect[i][j]) / (accuracy_values[j+1] - accuracy_values[j])
        norm_slope = slope * (accuracy_values[j] / accuracy_effect[i][j])
        ax.annotate(f'S={norm_slope:.2f}',
                    xy=((accuracy_values[j]+accuracy_values[j+1])/2,
                        (accuracy_effect[i][j]+accuracy_effect[i][j+1])/2),
                    xytext=(0, -12), textcoords='offset points',
                    color='#1f77b4', backgroundcolor='white', fontsize=8)

    for j in range(len(speed_values) - 1):
        slope = (speed_effect[i][j+1] - speed_effect[i][j]) / (speed_values[j+1] - speed_values[j])
        norm_slope = slope * (speed_values[j] / speed_effect[i][j])
        ax.annotate(f'S={norm_slope:.2f}',
                    xy=((speed_values[j]+speed_values[j+1])/2,
                        (speed_effect[i][j]+speed_effect[i][j+1])/2),
                    xytext=(-17, 10), textcoords='offset points',
                    color='#ff7f0e', backgroundcolor='white', fontsize=8)

    # 轴与标题
    ax.set_title(f'({chr(97+i)}) {titles[i]}', pad=6, fontweight='bold')
    ax.set_ylabel('Effectiveness' if i == 0 else '')
    ax.set_xlabel('platform-accuracy')
    ax.set_xticks(accuracy_values)
    ax.set_xticklabels(accuracy_values)

    # 顶部第二 X 轴（platform-speed）
    ax2 = ax.twiny()
    ax2.set_xlim(ax.get_xlim())
    ax2.set_xticks(speed_values)
    ax2.set_xticklabels(speed_values)
    ax2.set_xlabel('platform-speed')

# —— 合并图例：置于底部 —— 
handles, labels = [], []
for ax in axes:
    h, l = ax.get_legend_handles_labels()
    handles += h
    labels += l
# 去重保序
seen = set()
uniq = []
for h, l in zip(handles, labels):
    if l not in seen:
        uniq.append((h, l))
        seen.add(l)

fig.legend([h for h, _ in uniq], [l for _, l in uniq],
           loc='lower center', bbox_to_anchor=(0.5, -0.02),
           ncol=3, frameon=False, fontsize=9,
           title='Legend (S = normalized sensitivity coefficient)',
           title_fontsize=9)

# 为底部图例留白
plt.tight_layout(rect=(0, 0.08, 1, 1))

# 保存
plt.savefig('fig3_platform_sensitivity_horizontal_1x3.pdf',
            format='pdf', dpi=600, bbox_inches='tight')
plt.show()
plt.close()
print('fig3的pdf成功输出')
