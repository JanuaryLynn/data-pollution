import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns
from matplotlib.colors import LinearSegmentedColormap

# 创建图表
fig = plt.figure(figsize=(15, 12))
grid = plt.GridSpec(3, 3, height_ratios=[1, 3, 1], wspace=0.3, hspace=0.3)

# 定义数据
pollution_levels = [10, 30, 50]
legal_combinations = ['L1\n(S=30,R=5)', 'L2\n(S=60,R=5)', 'L3\n(S=90,R=5)', 'L4\n(S=60,R=1)', 'L5\n(S=60,R=10)']
platform_combinations = ['P1\n(A=70,S=75)', 'P2\n(A=80,S=75)', 'P3\n(A=90,S=75)', 'P4\n(A=80,S=50)', 'P5\n(A=80,S=100)']

# 创建复杂的效果矩阵（这里使用示例数据，您需要替换为实际数据）
# 行代表不同策略组合，列代表不同污染程度
legal_effectiveness = np.array([
    [18.273, 10.976, 7.314],    # L1
    [22.399, 13.875, 10.003],   # L2
    [28.453, 16.557, 12.524],   # L3
    [99.459, 39.492, 32.820],   # L4
    [19.477, 11.249, 7.594]     # L5
])

platform_effectiveness = np.array([
    [32.131, 26.281, 23.159],   # P1
    [34.238, 28.678, 25.849],   # P2
    [36.671, 31.077, 27.815],   # P3
    [28.053, 22.420, 18.889],   # P4
    [40.456, 34.935, 32.257]    # P5
])

# 计算策略优势矩阵
advantage_matrix = np.zeros((len(legal_combinations), len(platform_combinations), len(pollution_levels)))
for i in range(len(legal_combinations)):
    for j in range(len(platform_combinations)):
        for k in range(len(pollution_levels)):
            advantage_matrix[i, j, k] = legal_effectiveness[i, k] - platform_effectiveness[j, k]

# 创建自定义颜色映射，突出显示策略优势差异
colors = ["#053061", "#2166ac", "#4393c3", "#92c5de", "#d1e5f0", 
          "#f7f7f7", 
          "#fddbc7", "#f4a582", "#d6604d", "#b2182b", "#67001f"]
custom_cmap = LinearSegmentedColormap.from_list("custom", colors, N=256)

# 绘制热力矩阵
for p, pollution in enumerate(pollution_levels):
    ax = fig.add_subplot(grid[1, p])
    
    # 绘制热力图
    im = ax.imshow(advantage_matrix[:, :, p], cmap=custom_cmap, 
                   vmin=-50, vmax=50, aspect='auto')
    
    # 添加数值标签
    for i in range(len(legal_combinations)):
        for j in range(len(platform_combinations)):
            value = advantage_matrix[i, j, p]
            text_color = 'white' if abs(value) > 30 else 'black'
            ax.text(j, i, f'{value:.1f}', ha='center', va='center', 
                   color=text_color, fontsize=9, fontweight='bold')
    
    # 设置标签
    ax.set_xticks(np.arange(len(platform_combinations)))
    ax.set_yticks(np.arange(len(legal_combinations)))
    ax.set_xticklabels(platform_combinations, rotation=45, ha='right')
    ax.set_yticklabels(legal_combinations)
    ax.set_xlabel('Platform Self-regulation Strategies')
    if p == 0:
        ax.set_ylabel('Legal Intervention Strategies')
    ax.set_title(f'Pollution Level: {pollution}%')
    
    # 添加最优策略标记
    best_legal = np.argmax(legal_effectiveness[:, p])
    best_platform = np.argmax(platform_effectiveness[:, p])
    ax.plot(best_platform, best_legal, 'k*', markersize=15)

# 添加整体标题和图例
cbar_ax = fig.add_subplot(grid[0, :])
cbar = fig.colorbar(im, cax=cbar_ax, orientation='horizontal')
cbar.set_label('Strategy Advantage (Legal - Platform)', fontsize=12)
cbar_ax.set_title('Strategy-Context Adaptation Matrix', fontsize=16, pad=20)

# 添加解释性图例
explanation_ax = fig.add_subplot(grid[2, :])
explanation_ax.axis('off')
explanation_ax.text(0.5, 0.7, '* Indicates optimal parameter combination for each strategy', 
                   ha='center', fontsize=11)
explanation_ax.text(0.5, 0.4, 'Positive values (red): Legal intervention advantage', 
                   ha='center', fontsize=11)
explanation_ax.text(0.5, 0.1, 'Negative values (blue): Platform self-regulation advantage', 
                   ha='center', fontsize=11)

plt.tight_layout(rect=[0, 0, 1, 0.95])
plt.savefig('strategy_context_matrix.png', dpi=300, bbox_inches='tight')
plt.show()
