import matplotlib.pyplot as plt
import numpy as np
plt.style.use('default')
plt.rcParams.update({
    'font.family': 'serif',
    'font.size': 10,
    'axes.labelsize': 11
})
fig, axes = plt.subplots(1, 3, figsize=(18, 8))
pollution_levels = [10, 30, 50]
titles = ['Low Pollution (10%)', 'Medium Pollution (30%)', 'High Pollution (50%)']
base_values = [34.238, 28.678, 25.849]  # Effectiveness at 80% accuracy, 75 speed
parameters = ['platform-accuracy', 'platform-speed']
parameter_ranges = ['(70→90)', '(50→100)']
sensitivity_data = [
    # 10% pollution
    [[(-6.15), 7.11],  # platform-accuracy: (32.131-34.238)/34.238, (36.671-34.238)/34.238
     [(-18.07), 18.16]],  # platform-speed: (28.053-34.238)/34.238, (40.456-34.238)/34.238
    
    # 30% pollution
    [[(-8.36), 8.37],  # platform-accuracy: (26.281-28.678)/28.678, (31.077-28.678)/28.678
     [(-21.82), 21.82]],  # platform-speed: (22.420-28.678)/28.678, (34.935-28.678)/28.678
    
    # 50% pollution
    [[(-10.41), 7.61],  # platform-accuracy: (23.159-25.849)/25.849, (27.815-25.849)/25.849
     [(-26.93), 24.79]]  # platform-speed: (18.889-25.849)/25.849, (32.257-25.849)/25.849
]
colors = ['#d7191c', '#2c7bb6']
x_ranges = [[-30, 30], [-30, 30], [-30, 30]]  # Adjusted for platform data range
for i, pollution in enumerate(pollution_levels):
    ax = axes[i]
    # 按绝对影响排序
    impacts = [max(abs(sensitivity_data[i][j][0]), abs(sensitivity_data[i][j][1]))
               for j in range(len(parameters))]
    sorted_idx = np.argsort(impacts)[::-1]
    y_pos = np.arange(len(parameters))
   
    # 画柱子
    for j, idx in enumerate(sorted_idx):
        neg, pos = sensitivity_data[i][idx]
        ax.barh(y_pos[j], neg, 0.4, color=colors[0], alpha=0.7,
                label='Decrease' if (i == 0 and j == 0) else "")
        ax.barh(y_pos[j], pos, 0.4, color=colors[1], alpha=0.7,
                label='Increase' if (i == 0 and j == 0) else "")
    # y 轴标签
    ax.set_yticks(y_pos)
    ax.set_yticklabels([f"{parameters[idx]}\n{parameter_ranges[idx]}" for idx in sorted_idx])
    ax.set_xlim(x_ranges[i])
    ax.set_xlabel('% Change in Effectiveness', labelpad=10)
    if i == 0:
        ax.set_ylabel('Parameter')
        # 图例移出子图，放到左上角空白处
        ax.legend(loc='upper left', bbox_to_anchor=(-0.5, 1), framealpha=0.8)
    ax.set_title(titles[i], pad=10)
    ax.axvline(0, color='black', linestyle='-', alpha=0.5)
    ax.grid(True, linestyle='--', alpha=0.7, axis='x')
    # x 轴刻度
    ticks = []
    if x_ranges[i][0] <= -30:
        ticks.append(-30)
    if x_ranges[i][0] <= -20:
        ticks.append(-20)
    if x_ranges[i][0] <= -10:
        ticks.append(-10)
    ticks.append(0)
    if x_ranges[i][1] >= 10:
        ticks.append(10)
    if x_ranges[i][1] >= 20:
        ticks.append(20)
    if x_ranges[i][1] >= 30:
        ticks.append(30)
    ax.set_xticks(ticks)
    # 根据参数类型放置数值标签
    for j, idx in enumerate(sorted_idx):
        neg, pos = sensitivity_data[i][idx]
        # platform-speed 标签在柱子上方
        if parameters[idx] == 'platform-speed':
            y_off = 0.22
            va = 'bottom'
        else:  # platform-accuracy 标签在柱子下方
            y_off = -0.22
            va = 'top'
        # 负值 (Decrease) 标签
        ax.text(neg, y_pos[j] + y_off, f'{neg:.1f}%',
                va=va, ha='center', color='black', fontsize=6.5, fontweight='bold')
        # 正值 (Increase) 标签
        ax.text(pos, y_pos[j] + y_off, f'{pos:.1f}%',
                va=va, ha='center', color='black', fontsize=6.5, fontweight='bold')
# 调整间距
plt.tight_layout(rect=[0,0,1,0.92])
plt.suptitle('Sensitivity Analysis: Impact of Parameter Variations on Effectiveness',
             fontsize=16, y=0.98, fontweight='bold')  # 添加了fontweight='bold'参数
plt.savefig('platform_tornado_diagram.png', dpi=300, bbox_inches='tight')
plt.show()
