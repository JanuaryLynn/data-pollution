import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns

# Set professional style for publication
sns.set_theme(style="whitegrid")
plt.rcParams.update({
    'font.family': 'serif',
    'font.size': 11,
    'axes.labelsize': 12,
    'axes.titlesize': 14,
    'xtick.labelsize': 10,
    'ytick.labelsize': 10
})

# Create figure with subplots for different pollution levels
fig, axes = plt.subplots(1, 3, figsize=(18, 6), sharey=True)
pollution_levels = [10, 30, 50]
titles = ['Low Pollution (10%)', 'Medium Pollution (30%)', 'High Pollution (50%)']

# Reference values (no treatment)
no_treatment = [14.943, 8.575, 5.240]

# Legal strength sensitivity (when response-time = 5)
strength_values = [30, 60, 90]
strength_effect = [
    [18.273, 22.399, 28.453],  # 10% pollution
    [10.976, 13.875, 16.557],  # 30% pollution
    [7.314, 10.003, 12.524]    # 50% pollution
]

# Response time sensitivity (when strength = 60)
response_values = [1, 5, 10]
response_effect = [
    [99.459, 22.399, 19.477],  # 10% pollution
    [39.492, 13.875, 11.249],  # 30% pollution
    [32.820, 10.003, 7.594]    # 50% pollution
]

# Plot data
for i, pollution in enumerate(pollution_levels):
    ax = axes[i]
    
    # Plot strength sensitivity
    ax.plot(strength_values, strength_effect[i], 'o-',
            color='#1f77b4', linewidth=2, markersize=8,
            label='legal-strength (response-time=5)')
    
    # Plot response time sensitivity
    ax.plot(response_values, response_effect[i], 's-',
            color='#ff7f0e', linewidth=2, markersize=8,
            label='legal-response-time (strength=60)')
    
    # Reference line for no treatment
    ax.axhline(y=no_treatment[i], linestyle='--', color='gray',
               label='no treatment')
    
    # Annotate sensitivity slopes for strength
    for j in range(len(strength_values)-1):
        slope = (strength_effect[i][j+1] - strength_effect[i][j]) / (strength_values[j+1] - strength_values[j])
        norm_slope = slope * (strength_values[j] / strength_effect[i][j])
        ax.annotate(f'S={norm_slope:.2f}',
                    xy=((strength_values[j] + strength_values[j+1]) / 2,
                        (strength_effect[i][j] + strength_effect[i][j+1]) / 2),
                    xytext=(0, 15),  # shifted right and up
                    textcoords='offset points',
                    color='#1f77b4', backgroundcolor='white', fontsize=8)
    
    # Annotate sensitivity slopes for response time
    for j in range(len(response_values)-1):
        slope = (response_effect[i][j+1] - response_effect[i][j]) / (response_values[j+1] - response_values[j])
        norm_slope = slope * (response_values[j] / response_effect[i][j])
        ax.annotate(f'S={norm_slope:.2f}',
                    xy=((response_values[j] + response_values[j+1]) / 2,
                        (response_effect[i][j] + response_effect[i][j+1]) / 2),
                    xytext=(5, 8),  # shifted right and down
                    textcoords='offset points',
                    color='#ff7f0e', backgroundcolor='white', fontsize=8)
    
    # Labels and titles
    ax.set_xlabel('legal-strength')
    if i == 0:
        ax.set_ylabel('Effectiveness')
    # Increase pad to separate from secondary axis label
    ax.set_title(titles[i], pad=15)
    
    # Twin x-axis for response time
    ax2 = ax.twiny()
    ax2.set_xlim(ax.get_xlim())
    ax2.set_xticks(response_values)
    ax2.set_xticklabels(response_values)
    ax2.set_xlabel('legal-response-time')
    
    # Primary x-axis ticks
    ax.set_xticks(strength_values)
    ax.set_xticklabels(strength_values)

# Add global header (suptitle)
plt.subplots_adjust(top=0.85)
fig.suptitle('Parameter Sensitivity Analysis for Legal Intervention',
             fontsize=16, fontweight='bold')

# Add figure-level legend near header
handles, labels = axes[0].get_legend_handles_labels()
fig.legend(handles, labels,
           loc='upper left', bbox_to_anchor=(0.01, 0.92), framealpha=0.9,
           title='Legend (S = normalized sensitivity coefficient)', title_fontsize=10)

# Adjust layout to avoid overlap
plt.tight_layout(rect=[0, 0.0, 1, 0.82])

plt.savefig('parameter_sensitivity_curves_with_header.png', dpi=300, bbox_inches='tight')
plt.show()
