import matplotlib.pyplot as plt
import pandas as pd

# 设置字体和样式
plt.rcParams.update({
    "font.family": "Times New Roman",
    "text.usetex": False,
    "font.size": 9
})

# 表格内容准备（mathtext 风格符号）
data = {
    "Parameter": ["$p_0$", "$L$", r"$\tau_L$", r"$\alpha$", r"$n_p$"],
    "Default in code": ["—", "70", "5", "80", "75"],
    "Remark": [
        "Situation of initial pollution",
        "Mirrors escalating statutory sanctions",
        "Emergency to routine judicial timelines",
        "Moderation precision range",
        "Share of accounts audited per tick"
    ],
    "Unit": [
        "% of nodes", "% of nodes per tick", "ticks", "% of nodes per tick", "% of nodes per tick"
    ],
    "Scenario sweep": [
        "[10, 30, 50]", "[50, 70, 90]", "[1, 5, 10]", "[70, 80, 90]", "[50, 75, 100]"
    ]
}

df = pd.DataFrame(data)

# 绘图
fig, ax = plt.subplots(figsize=(8, 2.5))
ax.axis('off')

# 创建表格
table = ax.table(cellText=df.values,
                 colLabels=df.columns,
                 cellLoc='center',
                 loc='center')

# 样式设置
table.auto_set_font_size(False)
table.set_fontsize(9)

# 单独设置列宽
col_widths = [0.1, 0.15, 0.35, 0.17, 0.15]  # 参数、默认值列较窄，remark列更宽
for i in range(len(df.columns)):
    for j in range(len(df) + 1):  # 包括标题行
        table[j, i].set_width(col_widths[i])

table.scale(1.2, 1.5)

# 保存为 PDF
plt.savefig("table3_parameter_summary_adjusted.pdf", bbox_inches='tight', dpi=600)
plt.show()
