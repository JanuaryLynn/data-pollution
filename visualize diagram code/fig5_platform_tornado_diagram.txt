import matplotlib.pyplot as plt
import numpy as np

plt.rcParams.update({
    'font.family': 'Times New Roman',
    'font.size': 10,
    'axes.labelsize': 10,
    'axes.titlesize': 10,
    'xtick.labelsize': 9,
    'ytick.labelsize': 9
})

# Data
pollution_levels = [10, 30, 50]
titles = ['Low Pollution (10%)', 'Medium Pollution (30%)', 'High Pollution (50%)']
base_values = [34.238, 28.678, 25.849]
parameters = ['platform-accuracy', 'platform-speed']
parameter_ranges = ['(70↔90)', '(50↔100)']
sensitivity_data = [
    [[-6.15, 7.11], [-18.07, 18.16]],
    [[-8.36, 8.37], [-21.82, 21.82]],
    [[-10.41, 7.61], [-26.93, 24.79]]
]
colors = ['#d7191c', '#2c7bb6']
x_ranges = [[-30, 30], [-30, 30], [-30, 30]]

# Create vertical layout
fig, axes = plt.subplots(3, 1, figsize=(5.5, 12), sharex=False)

for i, pollution in enumerate(pollution_levels):
    ax = axes[i]
    impacts = [max(abs(sensitivity_data[i][j][0]), abs(sensitivity_data[i][j][1]))
               for j in range(len(parameters))]
    sorted_idx = np.argsort(impacts)[::-1]
    y_pos = np.arange(len(parameters))

    for j, idx in enumerate(sorted_idx):
        neg, pos = sensitivity_data[i][idx]
        ax.barh(y_pos[j], neg, 0.4, color=colors[0], alpha=0.7,
                label='Decrease' if (i == 0 and j == 0) else "")
        ax.barh(y_pos[j], pos, 0.4, color=colors[1], alpha=0.7,
                label='Increase' if (i == 0 and j == 0) else "")

    ax.set_yticks(y_pos)
    ax.set_yticklabels([f"{parameters[idx]}\n{parameter_ranges[idx]}" for idx in sorted_idx])
    ax.set_xlim(x_ranges[i])
    ax.set_xlabel('% Change in Effectiveness', labelpad=10)
    if i == 0:
        ax.legend(loc='upper center', bbox_to_anchor=(0.5, 1.35), ncol=2,
                  frameon=False, fontsize=9)

    ax.set_title(titles[i], pad=10)
    ax.axvline(0, color='black', linestyle='-', alpha=0.5)
    ax.grid(True, linestyle='--', alpha=0.7, axis='x')

    # X ticks
    ticks = []
    if x_ranges[i][0] <= -30:
        ticks.append(-30)
    if x_ranges[i][0] <= -20:
        ticks.append(-20)
    if x_ranges[i][0] <= -10:
        ticks.append(-10)
    ticks.append(0)
    if x_ranges[i][1] >= 10:
        ticks.append(10)
    if x_ranges[i][1] >= 20:
        ticks.append(20)
    if x_ranges[i][1] >= 30:
        ticks.append(30)
    ax.set_xticks(ticks)

    # Annotate bars
    for j, idx in enumerate(sorted_idx):
        neg, pos = sensitivity_data[i][idx]
        if parameters[idx] == 'platform-speed':
            y_off = 0.22
            va = 'bottom'
        else:
            y_off = -0.22
            va = 'top'
        ax.text(neg, y_pos[j] + y_off, f'{neg:.1f}%',
                va=va, ha='center', color='black', fontsize=6.5, fontweight='bold')
        ax.text(pos, y_pos[j] + y_off, f'{pos:.1f}%',
                va=va, ha='center', color='black', fontsize=6.5, fontweight='bold')

    # Subfigure label
    ax.text(0.5, -0.3, f'({chr(97+i)}) {titles[i]}',
            transform=ax.transAxes, ha='center', va='top', fontsize=9, fontweight='bold')

# Adjust layout
plt.subplots_adjust(hspace=0.7)

# Save as PDF
plt.savefig('fig5_platform_tornado_sensitivity_vertical.pdf', format='pdf', dpi=600, bbox_inches='tight')
plt.show()
plt.close()
print('fig5的pdf成功输出')
